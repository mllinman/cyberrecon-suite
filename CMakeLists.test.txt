cmake_minimum_required(VERSION 3.16)
project(CyberReconSuiteTest)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6 components - minimal for testing
find_package(Qt6 QUIET COMPONENTS 
    Core 
    Widgets 
)

if(Qt6_FOUND)
    message(STATUS "Qt6 found, building full application")
    
    # Enable automatic MOC, UIC, and RCC
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTOUIC ON)
    set(CMAKE_AUTORCC ON)

    # Include directories
    include_directories(src)

    # Simple test with minimal files
    set(SOURCES
        src/main.cpp
    )
    
    # Create test executable
    add_executable(CyberReconSuiteTest ${SOURCES})
    
    # Link Qt6 libraries
    target_link_libraries(CyberReconSuiteTest PRIVATE
        Qt6::Core
        Qt6::Widgets
    )
    
    # Set target properties
    set_target_properties(CyberReconSuiteTest PROPERTIES
        WIN32_EXECUTABLE TRUE
        OUTPUT_NAME "CyberReconSuite"
    )
    
else()
    message(STATUS "Qt6 not found, building minimal console test")
    
    # Create minimal test executable
    add_executable(CyberReconSuiteTest 
        test/minimal_test.cpp
    )
    
endif()

# Install rules
install(TARGETS CyberReconSuiteTest
    RUNTIME DESTINATION bin
)
